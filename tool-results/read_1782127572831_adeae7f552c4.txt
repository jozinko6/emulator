    71→    71→/** Source type label pre ImportJobRecord.sourceType. */
    72→    72→function sourceTypeFor(format: string): string {
    73→    73→  if (format === "zip" || format === "rar" || format === "7z") return format;
    74→    74→  if (format === "jsdos") return "jsdos";
    75→    75→  return "local-file";
    76→    76→}
    77→    77→
    78→    78→/** Vytvorí nový ImportJobRecord. */
    79→    79→function createJob(fileName: string, fileSize: number, sourceType: string): ImportJobRecord {
    80→    80→  const now = Date.now();
    81→    81→  return {
    82→    82→    id: uuid(),
    83→    83→    status: "selecting",
    84→    84→    sourceType,
    85→    85→    fileName,
    86→    86→    totalBytes: fileSize,
    87→    87→    processedBytes: 0,
    88→    88→    extractedBytes: 0,
    89→    89→    fileCount: 0,
    90→    90→    processedFiles: 0,
    91→    91→    currentStep: "Pripravený na spracovanie",
    92→    92→    warnings: [],
    93→    93→    createdAt: now,
    94→    94→    updatedAt: now,
    95→    95→  };
    96→    96→}

    98→    98→export function ImportWizard() {
    99→    99→  const {
   100→   100→    currentJob,
   101→   101→    setJob,
   102→   102→    updateJob,
   103→   103→    finalizeJob,
   104→   104→    legalConfirmed,
   105→   105→    setLegalConfirmed,
   106→   106→    reset,
   107→   107→  } = useImportStore();
   108→   108→  const addGame = useLibraryStore((s) => s.addGame);
   109→   109→  const { toast } = useToast();
   110→   110→
   111→   111→  // Lokálny stav
   112→   112→  const [legalChecked, setLegalChecked] = useState(false);
   113→   113→  const [selectedPlatform, setSelectedPlatform] = useState<EmulatorPlatform | null>(null);
   114→   114→  const [detectionResult, setDetectionResult] = useState<DetectionResult | null>(null);
   115→   115→  const [extractedEntries, setExtractedEntries] = useState<ArchiveEntry[]>([]);
   116→   116→
   117→   117→  // Refy pre async operácie
   118→   118→  const abortRef = useRef<AbortController | null>(null);
   119→   119→  const workerRef = useRef<Worker | null>(null);
   120→   120→  const gameIdRef = useRef<string>("");
   121→   121→  const writeQueueRef = useRef<
   122→   122→    Array<{ path: string; data: Uint8Array; size: number }>
   123→   123→  >([]);
   124→   124→  const writePromiseRef = useRef<Promise<void> | null>(null);
   125→   125→  const extractedEntriesRef = useRef<ArchiveEntry[]>([]);
   126→   126→  const mainFileRef = useRef<string>("");
   127→   127→
   128→   128→  // Cleanup pri unmount
   129→   129→  useEffect(() => {
   130→   130→    return () => {
   131→   131→      abortRef.current?.abort();
   132→   132→      if (workerRef.current) {
   133→   133→        workerRef.current.terminate();
   134→   134→        workerRef.current = null;
   135→   135→      }
   136→   136→    };
   137→   137→  }, []);
   138→   138→
   139→   139→  /**
   140→   140→   * Začne import — volá sa z DropZone po výbere súborov.
   141→   141→   * Spracuje len prvý súbor (ostatné preskočí s upozornením).
   142→   142→   */
   143→   143→  const handleFilesSelected = useCallback(
   144→   144→    (files: File[]) => {
   145→   145→      if (files.length === 0) return;
   146→   146→      if (!legalConfirmed) return;
   147→   147→
   148→   148→      const file = files[0];
   149→   149→      const fmt = detectFormat(file.name);
   150→   150→      if (!isSupported(fmt)) {
   151→   151→        toast({
   152→   152→          title: "Nepodporovaný súbor",
   153→   153→          description: `Súbor ${file.name} nie je podporovaného formátu.`,
   154→   154→          variant: "destructive",
   155→   155→        });
   156→   156→        return;
   157→   157→      }
   158→   158→
   159→   159→      if (files.length > 1) {
   160→   160→        toast({
   161→   161→          title: "Viac súborov",
   162→   162→          description: `Bude sa importovať iba ${file.name}. Ostatné presuňte neskôr.`,
   163→   163→        });
   164→   164→      }
   165→   165→
   166→   166→      const sourceType = sourceTypeFor(fmt);
   167→   167→      const job = createJob(file.name, file.size, sourceType);
   168→   168→      setJob(job);
   169→   169→      setExtractedEntries([]);
   170→   170→      setDetectionResult(null);
   171→   171→      setSelectedPlatform(null);
   172→   172→      extractedEntriesRef.current = [];
   173→   173→      mainFileRef.current = "";
   174→   174→      writeQueueRef.current = [];
   175→   175→      writePromiseRef.current = null;
   176→   176→
   177→   177→      // Spustenie pipeline
   178→   178→      void runPipeline(file, fmt, job);
   179→   179→    },
   180→   180→     
   181→   181→    [legalConfirmed, setJob, toast]
   182→   182→  );
   183→   183→
   184→   184→  /**
   185→   185→   * Hlavná pipeline — spúšťa jednotlivé fázy importu.
   186→   186→   * Je izolovaná do jednej async funkcie kvôli čitateľnosti.
   187→   187→   */
   188→   188→  const runPipeline = useCallback(
   189→   189→    async (file: File, format: ImportableFormat, initialJob: ImportJobRecord) => {
   190→   190→      const jobId = initialJob.id;
   191→   191→      const gameId = uuid();
   192→   192→      gameIdRef.current = gameId;
   193→   193→      const abort = new AbortController();
   194→   194→      abortRef.current = abort;
   195→   195→
   196→   196→      try {
   197→   197→        // === READING ===
   198→   198→        updateJob({ status: "reading", currentStep: "Čítanie súboru..." });
   199→   199→
   200→   200→        // Pre archívy potrebujeme načítať dáta (pre ZIP) alebo File (pre RAR/7z)
   201→   201→        // Streamer sa postará o OPFS neskôr.
   202→   202→        let entries: ArchiveEntry[] = [];
   203→   203→
   204→   204→        if (isArchive(format)) {
   205→   205→          // === VALIDATING ===
   206→   206→          updateJob({ status: "validating", currentStep: "Validácia archívu..." });
   207→   207→
   208→   208→          // Pre ZIP potrebujeme Uint8Array — ak je súbor veľký, varujeme.
   209→   209→          // Pre RAR/7z pošleme File priamo do workera.
   210→   210→          const workerPayload: File | ArrayBuffer =
   211→   211→            format === "zip"
   212→   212→              ? await file.arrayBuffer()
   213→   213→              : file;
   214→   214→
   215→   215→          // Spustenie workera pre list+extract
   216→   216→          entries = await runArchiveWorker(
   217→   217→            format as "zip" | "rar" | "7z",
   218→   218→            workerPayload,
   219→   219→            file.name,
   220→   220→            abort.signal
   221→   221→          );
   222→   222→
   223→   223→          // Skontrolujeme bezpečnosť archívu
   224→   224→          const summary = summarizeArchive(entries);
   225→   225→          const validation = validateSummary(summary, DEFAULT_ARCHIVE_LIMITS);
   226→   226→          if (!validation.ok) {
   227→   227→            // Pridáme varovania do job-u
   228→   228→            updateJob({ warnings: [...(currentJob?.warnings ?? []), ...validation.reasons] });
   229→   229→            // Ak je to archívna bomba alebo path traversal, zastavíme
   230→   230→            const critical = validation.reasons.some(
   231→   231→              (r) => r.includes("archívnu bombu") || r.includes("path traversal")
   232→   232→            );
   233→   233→            if (critical) {
   234→   234→              throw new RetroCloudError(
   235→   235→                "ARCHIVE_BOMB_DETECTED",
   236→   236→                "Archív bol zamietnutý z bezpečnostných dôvodov."
   237→   237→              );
   238→   238→            }
   239→   239→          }
   240→   240→
   241→   241→          // Skontrolujeme nebezpečné cesty
   242→   242→          const unsafePaths = checkEntriesForUnsafePaths(entries);
   243→   243→          if (unsafePaths.length > 0) {
   244→   244→            updateJob({
   245→   245→              warnings: [
   246→   246→                ...(currentJob?.warnings ?? []),
   247→   247→                ...unsafePaths.slice(0, 50).map((p) => `Nebezpečná cesta: ${p}`),
   248→   248→              ],
   249→   249→            });
   250→   250→            // Filtrujeme von nebezpečné cesty
   251→   251→            entries = entries.filter((e) => {
   252→   252→              try {
   253→   253→                normalizePath(e.path);
   254→   254→                return true;
   255→   255→              } catch {
   256→   256→                return false;
   257→   257→              }
   258→   258→            });
   259→   259→          }
   260→   260→
   261→   261→          // === EXTRACTING (streaming do OPFS) ===
   262→   262→          updateJob({
   263→   263→            status: "extracting",
   264→   264→            currentStep: "Rozbaľovanie archívu a zápis do OPFS...",
   265→   265→            fileCount: summary.totalFiles,
   266→   266→            totalBytes: summary.totalUncompressedSize,
   267→   267→          });
   268→   268→
   269→   269→          // Worker už rozbalil a poslal entries — writes už prebehli v runArchiveWorker.
   270→   270→          extractedEntriesRef.current = entries;
   271→   271→          setExtractedEntries(entries);
   272→   272→        } else {
   273→   273→          // Voľný súbor — žiadny archív na rozbalenie
   274→   274→          updateJob({
   275→   275→            status: "validating",
   276→   276→            currentStep: "Kontrola súboru...",
   277→   277→            fileCount: 1,
   278→   278→            totalBytes: file.size,
   279→   279→          });
   280→   280→
   281→   281→          if (isStreamRequired(file.size)) {
   282→   282→            // Veľký súbor — streamujeme priamo v "storing" fáze
   283→   283→            updateJob({
   284→   284→              warnings: [
   285→   285→                ...(currentJob?.warnings ?? []),
   286→   286→                `Súbor je väčší ako 256 MB (${(file.size / 1024 / 1024).toFixed(0)} MB) — bude sa streamovať.`,
   287→   287→              ],
   288→   288→            });
   289→   289→          }
   290→   290→
   291→   291→          // Vytvoríme "pseudo-entry" pre detekciu
   292→   292→          const pseudoEntry: ArchiveEntry = {
   293→   293→            path: file.name,
   294→   294→            size: file.size,
   295→   295→            isDirectory: false,
   296→   296→          };
   297→   297→          entries = [pseudoEntry];
   298→   298→          extractedEntriesRef.current = entries;
   299→   299→          setExtractedEntries(entries);
   300→   300→
   301→   301→          // === STORING (streaming do OPFS) ===
   302→   302→          updateJob({
   303→   303→            status: "storing",
   304→   304→            currentStep: "Zápis súboru do OPFS...",
   305→   305→          });
   306→   306→
   307→   307→          const opfsPath = gameFilePath(gameId, file.name);
   308→   308→          const result = await streamFileToOpfs(
   309→   309→            file,
   310→   310→            opfsPath,
   311→   311→            (written) => {
   312→   312→              updateJob({
   313→   313→                processedBytes: written,
   314→   314→                extractedBytes: written,
   315→   315→              });
   316→   316→            },
   317→   317→            abort.signal
   318→   318→          );
   319→   319→
   320→   320→          updateJob({
   321→   321→            processedBytes: result.size,
   322→   322→            extractedBytes: result.size,
   323→   323→            processedFiles: 1,
   324→   324→          });
   325→   325→
   326→   326→          mainFileRef.current = file.name;
   327→   327→        }
   328→   328→
   329→   329→        // Vyberieme main file pre detekciu
   330→   330→        const mainFile = pickMainFile(extractedEntriesRef.current);
   331→   331→        mainFileRef.current = mainFile;
   332→   332→
   333→   333→        // === DETECTING ===
   334→   334→        updateJob({
   335→   335→          status: "detecting",
   336→   336→          currentStep: "Detekcia platformy...",
   337→   337→        });
   338→   338→
   339→   339→        const detection = await runDetectionWorker({
   340→   340→          fileName: mainFile,
   341→   341→          fileSize: extractedEntriesRef.current.find((e) => e.path === mainFile)?.size ?? 0,
   342→   342→        });
   343→   343→        setDetectionResult(detection);
   344→   344→
   345→   345→        // === AWAITING USER SELECTION (ak potrebné) ===
   346→   346→        if (detection.requiresUserSelection || !detection.platform) {
   347→   347→          updateJob({
   348→   348→            status: "awaiting-user-selection",
   349→   349→            currentStep: "Vyberte platformu hry",
   350→   350→          });
   351→   351→          // Tu sa pipeline pozastaví — pokračuje sa až po kliknutí na tlačidlo
   352→   352→          return;
   353→   353→        }
   354→   354→
   355→   355→        // Pokračujeme priamo na finalization
   356→   356→        await finalizeImport(detection.platform);
   357→   357→
   358→   358→         
   359→   359→      } catch (e: any) {
   360→   360→        if (abort.signal.aborted) {
   361→   361→          updateJob({
   362→   362→            status: "cancelled",
   363→   363→            currentStep: "Import bol zrušený",
   364→   364→          });
   365→   365→          // Cleanup OPFS
   366→   366→          await cleanupPartialImport(gameId).catch(() => undefined);
   367→   367→          return;
   368→   368→        }
   369→   369→        const message = e instanceof Error ? e.message : String(e);
   370→   370→        const code = e instanceof RetroCloudError ? e.code : "UNKNOWN_ERROR";
   371→   371→        updateJob({
   372→   372→          status: "error",
   373→   373→          currentStep: "Import zlyhal",
   374→   374→          error: `${message}${code !== "UNKNOWN_ERROR" ? ` (kód: ${code})` : ""}`,
   375→   375→        });
   376→   376→        // Cleanup OPFS pri chybe
   377→   377→        await cleanupPartialImport(gameId).catch(() => undefined);
   378→   378→        toast({
   379→   379→          title: "Import zlyhal",
   380→   380→          description: message,
   381→   381→          variant: "destructive",
   382→   382→        });
   383→   383→      } finally {
   384→   384→        if (workerRef.current) {
   385→   385→          workerRef.current.terminate();
   386→   386→          workerRef.current = null;
   387→   387→        }
   388→   388→      }
   389→   389→    },
   390→   390→     
   391→   391→    [updateJob, currentJob, setJob, toast]
   392→   392→  );
   393→   393→
   394→   394→  /**
   395→   395→   * Spustí archive worker pre rozbalenie archívu.
   396→   396→   * Vráti zoznam extrahovaných záznamov a súčasne streamuje dáta do OPFS.
   397→   397→   */
   398→   398→  const runArchiveWorker = useCallback(
   399→   399→    (
   400→   400→      format: "zip" | "rar" | "7z",
   401→   401→      payload: File | ArrayBuffer,
   402→   402→      fileName: string,
   403→   403→      signal: AbortSignal
   404→   404→    ): Promise<ArchiveEntry[]> => {
   405→   405→      return new Promise<ArchiveEntry[]>((resolve, reject) => {
   406→   406→        const worker = new Worker(
   407→   407→          new URL("../../workers/archive.worker.ts", import.meta.url),
   408→   408→          { type: "module" }
   409→   409→        );
   410→   410→        workerRef.current = worker;
   411→   411→        const entries: ArchiveEntry[] = [];
   412→   412→        let totalFiles = 0;
   413→   413→        let totalBytes = 0;
   414→   414→        let processedFiles = 0;
   415→   415→        let processedBytes = 0;
   416→   416→        let resolved = false;
   417→   417→
   418→   418→        const finish = (err?: Error) => {
   419→   419→          if (resolved) return;
   420→   420→          resolved = true;
   421→   421→          worker.onmessage = null;
   422→   422→          if (err) {
   423→   423→            reject(err);
   424→   424→          } else {
   425→   425→            resolve(entries);
   426→   426→          }
   427→   427→        };
   428→   428→
   429→   429→        worker.onmessage = (e: MessageEvent<ArchiveWorkerResponse>) => {
   430→   430→          const msg = e.data;
   431→   431→          if (!msg || typeof msg !== "object" || !("type" in msg)) return;
   432→   432→
   433→   433→          switch (msg.type) {
   434→   434→            case "progress": {
   435→   435→              totalFiles = msg.totalFiles;
   436→   436→              totalBytes = msg.totalBytes;
   437→   437→              updateJob({
   438→   438→                processedBytes: msg.processedBytes,
   439→   439→                totalBytes,
   440→   440→                processedFiles: msg.processedFiles,
   441→   441→                fileCount: totalFiles,
   442→   442→                extractedBytes: msg.processedBytes,
   443→   443→              });
   444→   444→              break;
   445→   445→            }
   446→   446→            case "warning": {
   447→   447→              const warning = `${msg.path} — ${msg.reason}`;
   448→   448→              updateJob({
   449→   449→                warnings: [...(useImportStore.getState().currentJob?.warnings ?? []), warning],
   450→   450→              });
   451→   451→              break;
   452→   452→            }
   453→   453→            case "entry": {
   454→   454→              // Pridáme do fronty na OPFS zápis
   455→   455→              writeQueueRef.current.push({
   456→   456→                path: msg.path,
   457→   457→                data: msg.data,
   458→   458→                size: msg.size,
   459→   459→              });
   460→   460→              entries.push({
   461→   461→                path: msg.path,
   462→   462→                size: msg.size,
   463→   463→                isDirectory: false,
   464→   464→              });
   465→   465→              processedFiles += 1;
   466→   466→              processedBytes += msg.size;
   467→   467→              // Spustíme flush ak nie je aktívny
   468→   468→              if (!writePromiseRef.current) {
   469→   469→                writePromiseRef.current = flushWriteQueue(
   470→   470→                  gameIdRef.current,
   471→   471→                  abortRef.current?.signal ?? new AbortController().signal,
   472→   472→                  (written) => {
   473→   473→                    updateJob({ processedBytes: written });
   474→   474→                  }
   475→   475→                ).finally(() => {
   476→   476→                  writePromiseRef.current = null;
   477→   477→                });
   478→   478→              }
   479→   479→              break;
   480→   480→            }
   481→   481→            case "done": {
   482→   482→              // Počkáme na vyprázdnenie write fronty
   483→   483→              const wait = () => {
   484→   484→                if (writePromiseRef.current) {
   485→   485→                  writePromiseRef.current.then(wait, wait);
   486→   486→                } else {
   487→   487→                  finish();
   488→   488→                }
   489→   489→              };
   490→   490→              wait();
   491→   491→              break;
   492→   492→            }
   493→   493→            case "cancelled": {
   494→   494→              finish(new RetroCloudError("UNKNOWN_ERROR", "Rozbaľovanie zrušené."));
   495→   495→              break;
   496→   496→            }
   497→   497→            case "error": {
   498→   498→              finish(
   499→   499→                new RetroCloudError(
   500→   500→                  (msg as { code: string }).code as RetroCloudError["code"],

   507→   507→              // Ignorujeme — používame entry správy

   523→   523→        if (payload instanceof ArrayBuffer) {

   526→   526→          worker.postMessage(request);
   527→   527→        }
   528→   528→
   529→   529→        // Abort handler
   530→   530→        signal.addEventListener("abort", () => {
   531→   531→          if (workerRef.current) {

   543→   543→   * Sériovo — aby sme nezaťažili OPFS paralelnými zápismi.
   544→   544→   */
   545→   545→  const flushWriteQueue = useCallback(
   546→   546→    async (
   547→   547→      gameId: string,
   548→   548→      signal: AbortSignal,
   549→   549→      onProgress: (written: number) => void
   550→   550→    ): Promise<void> => {
   551→   551→      let written = 0;
   552→   552→      while (writeQueueRef.current.length > 0) {
   553→   553→        if (signal.aborted) {
   554→   554→          throw new RetroCloudError("UNKNOWN_ERROR", "Zápis zrušený.");
   555→   555→        }
   556→   556→        const item = writeQueueRef.current.shift()!;
   557→   557→        const opfsPath = gameFilePath(gameId, item.path);
   558→   558→        // Kopírujeme dáta do čistého ArrayBufferu, aby sme vyhli TS problémom
   559→   559→        // s Uint8Array<ArrayBufferLike> v Blob konštruktore.
   560→   560→        const buffer = new ArrayBuffer(item.data.byteLength);
   561→   561→        new Uint8Array(buffer).set(item.data);
   562→   562→        const blob = new Blob([buffer]);
   563→   563→        const result = await streamBlobToOpfs(blob, opfsPath, (w) => {
   564→   564→          onProgress(written + w);
   565→   565→        }, signal);
   566→   566→        written += result.size;
   567→   567→        onProgress(written);
   568→   568→      }
   569→   569→    },
   570→   570→    []
