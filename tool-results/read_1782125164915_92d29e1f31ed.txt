   237→   237→
   238→   238→Použi koncept podobný tomuto:
   239→   239→
   240→   240→export type EmulatorPlatform = "dos" | "ps1" | "ps2";
   241→   241→
   242→   242→export type EmulatorLifecycleState =
   243→   243→  | "idle"
   244→   244→  | "initializing"
   245→   245→  | "loading"
   246→   246→  | "ready"
   247→   247→  | "running"
   248→   248→  | "paused"
   249→   249→  | "stopping"
   250→   250→  | "error"
   251→   251→  | "destroyed";
   252→   252→
   253→   253→export interface EmulatorInputEvent {
   254→   254→  type:
   255→   255→    | "button-down"
   256→   256→    | "button-up"
   257→   257→    | "axis"
   258→   258→    | "key-down"
   259→   259→    | "key-up"
   260→   260→    | "pointer";
   261→   261→  control: string;
   262→   262→  value?: number;
   263→   263→  x?: number;
   264→   264→  y?: number;

   270→   270→
   271→   271→  initialize(container: HTMLElement): Promise<void>;
   272→   272→  loadGame(game: ImportedGame): Promise<void>;
   273→   273→  start(): Promise<void>;
   274→   274→  pause(): Promise<void>;
   275→   275→  resume(): Promise<void>;
   276→   276→  reset(): Promise<void>;
   277→   277→
   278→   278→  sendInput(event: EmulatorInputEvent): void;
   279→   279→
   280→   280→  saveState(slot: number): Promise<StoredSaveState>;

   283→   283→
   284→   284→  setVolume(volume: number): void;
   285→   285→  setMuted(muted: boolean): void;
   286→   286→
   287→   287→  enterFullscreen(): Promise<void>;
   288→   288→  exitFullscreen(): Promise<void>;
   289→   289→
   290→   290→  getPerformanceStats(): EmulatorPerformanceStats;
   291→   291→  destroy(): Promise<void>;
   292→   292→
   293→   293→  subscribe(
   294→   294→    listener: (event: EmulatorEvent) => void
   295→   295→  ): () => void;
