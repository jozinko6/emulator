    71→/** Source type label pre ImportJobRecord.sourceType. */
    72→function sourceTypeFor(format: string): string {
    73→  if (format === "zip" || format === "rar" || format === "7z") return format;
    74→  if (format === "jsdos") return "jsdos";
    75→  return "local-file";
    76→}

    78→/** Vytvorí nový ImportJobRecord. */
    79→function createJob(fileName: string, fileSize: number, sourceType: string): ImportJobRecord {
    80→  const now = Date.now();
    81→  return {
    82→    id: uuid(),
    83→    status: "selecting",
    84→    sourceType,
    85→    fileName,
    86→    totalBytes: fileSize,
    87→    processedBytes: 0,
    88→    extractedBytes: 0,
    89→    fileCount: 0,
    90→    processedFiles: 0,
    91→    currentStep: "Pripravený na spracovanie",
    92→    warnings: [],
    93→    createdAt: now,
    94→    updatedAt: now,
    95→  };
    96→}

    98→export function ImportWizard() {
    99→  const {
   100→    currentJob,
   101→    setJob,
   102→    updateJob,
   103→    finalizeJob,
   104→    legalConfirmed,
   105→    setLegalConfirmed,
   106→    reset,
   107→  } = useImportStore();
   108→  const addGame = useLibraryStore((s) => s.addGame);
   109→  const { toast } = useToast();
   110→
   111→  // Lokálny stav
   112→  const [legalChecked, setLegalChecked] = useState(false);
   113→  const [selectedPlatform, setSelectedPlatform] = useState<EmulatorPlatform | null>(null);
   114→  const [detectionResult, setDetectionResult] = useState<DetectionResult | null>(null);
   115→  const [extractedEntries, setExtractedEntries] = useState<ArchiveEntry[]>([]);
   116→
   117→  // Refy pre async operácie
   118→  const abortRef = useRef<AbortController | null>(null);
   119→  const workerRef = useRef<Worker | null>(null);
   120→  const gameIdRef = useRef<string>("");
   121→  const writeQueueRef = useRef<
   122→    Array<{ path: string; data: Uint8Array; size: number }>
   123→  >([]);
   124→  const writePromiseRef = useRef<Promise<void> | null>(null);
   125→  const extractedEntriesRef = useRef<ArchiveEntry[]>([]);
   126→  const mainFileRef = useRef<string>("");
   127→
   128→  // Cleanup pri unmount
   129→  useEffect(() => {
   130→    return () => {
   131→      abortRef.current?.abort();
   132→      if (workerRef.current) {
   133→        workerRef.current.terminate();
   134→        workerRef.current = null;
   135→      }
   136→    };
   137→  }, []);
   138→
   139→  /**
   140→   * Začne import — volá sa z DropZone po výbere súborov.
   141→   * Spracuje len prvý súbor (ostatné preskočí s upozornením).
   142→   */
   143→  const handleFilesSelected = useCallback(
   144→    (files: File[]) => {
   145→      if (files.length === 0) return;
   146→      if (!legalConfirmed) return;
   147→
   148→      const file = files[0];
   149→      const fmt = detectFormat(file.name);
   150→      if (!isSupported(fmt)) {
   151→        toast({
   152→          title: "Nepodporovaný súbor",
   153→          description: `Súbor ${file.name} nie je podporovaného formátu.`,
   154→          variant: "destructive",
   155→        });
   156→        return;
   157→      }
   158→
   159→      if (files.length > 1) {
   160→        toast({
   161→          title: "Viac súborov",
   162→          description: `Bude sa importovať iba ${file.name}. Ostatné presuňte neskôr.`,
   163→        });
   164→      }
   165→
   166→      const sourceType = sourceTypeFor(fmt);
   167→      const job = createJob(file.name, file.size, sourceType);
   168→      setJob(job);
   169→      setExtractedEntries([]);
   170→      setDetectionResult(null);
   171→      setSelectedPlatform(null);
   172→      extractedEntriesRef.current = [];
   173→      mainFileRef.current = "";
   174→      writeQueueRef.current = [];
   175→      writePromiseRef.current = null;
   176→
   177→      // Spustenie pipeline
   178→      void runPipeline(file, fmt, job);
   179→    },
   180→     
   181→    [legalConfirmed, setJob, toast]
   182→  );
   183→
   184→  /**
   185→   * Hlavná pipeline — spúšťa jednotlivé fázy importu.
   186→   * Je izolovaná do jednej async funkcie kvôli čitateľnosti.
   187→   */
   188→  const runPipeline = useCallback(
   189→    async (file: File, format: ImportableFormat, initialJob: ImportJobRecord) => {
   190→      const jobId = initialJob.id;
   191→      const gameId = uuid();
   192→      gameIdRef.current = gameId;
   193→      const abort = new AbortController();
   194→      abortRef.current = abort;
   195→
   196→      try {
   197→        // === READING ===
   198→        updateJob({ status: "reading", currentStep: "Čítanie súboru..." });
   199→
   200→        // Pre archívy potrebujeme načítať dáta (pre ZIP) alebo File (pre RAR/7z)
   201→        // Streamer sa postará o OPFS neskôr.
   202→        let entries: ArchiveEntry[] = [];
   203→
   204→        if (isArchive(format)) {
   205→          // === VALIDATING ===
   206→          updateJob({ status: "validating", currentStep: "Validácia archívu..." });
   207→
   208→          // Pre ZIP potrebujeme Uint8Array — ak je súbor veľký, varujeme.
   209→          // Pre RAR/7z pošleme File priamo do workera.
   210→          const workerPayload: File | ArrayBuffer =
   211→            format === "zip"
   212→              ? await file.arrayBuffer()
   213→              : file;
   214→
   215→          // Spustenie workera pre list+extract
   216→          entries = await runArchiveWorker(
   217→            format as "zip" | "rar" | "7z",
   218→            workerPayload,
   219→            file.name,
   220→            abort.signal
   221→          );
   222→
   223→          // Skontrolujeme bezpečnosť archívu
   224→          const summary = summarizeArchive(entries);
   225→          const validation = validateSummary(summary, DEFAULT_ARCHIVE_LIMITS);
   226→          if (!validation.ok) {
   227→            // Pridáme varovania do job-u
   228→            updateJob({ warnings: [...(currentJob?.warnings ?? []), ...validation.reasons] });
   229→            // Ak je to archívna bomba alebo path traversal, zastavíme
   230→            const critical = validation.reasons.some(
   231→              (r) => r.includes("archívnu bombu") || r.includes("path traversal")
   232→            );
   233→            if (critical) {
   234→              throw new RetroCloudError(
   235→                "ARCHIVE_BOMB_DETECTED",
   236→                "Archív bol zamietnutý z bezpečnostných dôvodov."
   237→              );
   238→            }
   239→          }
   240→
   241→          // Skontrolujeme nebezpečné cesty
   242→          const unsafePaths = checkEntriesForUnsafePaths(entries);
   243→          if (unsafePaths.length > 0) {
   244→            updateJob({
   245→              warnings: [
   246→                ...(currentJob?.warnings ?? []),
   247→                ...unsafePaths.slice(0, 50).map((p) => `Nebezpečná cesta: ${p}`),
   248→              ],
   249→            });
   250→            // Filtrujeme von nebezpečné cesty
   251→            entries = entries.filter((e) => {
   252→              try {
   253→                normalizePath(e.path);
   254→                return true;
   255→              } catch {
   256→                return false;
   257→              }
   258→            });
   259→          }
   260→
   261→          // === EXTRACTING (streaming do OPFS) ===
   262→          updateJob({
   263→            status: "extracting",
   264→            currentStep: "Rozbaľovanie archívu a zápis do OPFS...",
   265→            fileCount: summary.totalFiles,
   266→            totalBytes: summary.totalUncompressedSize,
   267→          });
   268→
   269→          // Worker už rozbalil a poslal entries — writes už prebehli v runArchiveWorker.
   270→          extractedEntriesRef.current = entries;
   271→          setExtractedEntries(entries);
   272→        } else {
   273→          // Voľný súbor — žiadny archív na rozbalenie
   274→          updateJob({
   275→            status: "validating",
   276→            currentStep: "Kontrola súboru...",
   277→            fileCount: 1,
   278→            totalBytes: file.size,
   279→          });
   280→
   281→          if (isStreamRequired(file.size)) {
   282→            // Veľký súbor — streamujeme priamo v "storing" fáze
   283→            updateJob({
   284→              warnings: [
   285→                ...(currentJob?.warnings ?? []),
   286→                `Súbor je väčší ako 256 MB (${(file.size / 1024 / 1024).toFixed(0)} MB) — bude sa streamovať.`,
   287→              ],
   288→            });
   289→          }
   290→
   291→          // Vytvoríme "pseudo-entry" pre detekciu
   292→          const pseudoEntry: ArchiveEntry = {
   293→            path: file.name,
   294→            size: file.size,
   295→            isDirectory: false,
   296→          };
   297→          entries = [pseudoEntry];
   298→          extractedEntriesRef.current = entries;
   299→          setExtractedEntries(entries);
   300→
   301→          // === STORING (streaming do OPFS) ===
   302→          updateJob({
   303→            status: "storing",
   304→            currentStep: "Zápis súboru do OPFS...",
   305→          });
   306→
   307→          const opfsPath = gameFilePath(gameId, file.name);
   308→          const result = await streamFileToOpfs(
   309→            file,
   310→            opfsPath,
   311→            (written) => {
   312→              updateJob({
   313→                processedBytes: written,
   314→                extractedBytes: written,
   315→              });
   316→            },
   317→            abort.signal
   318→          );
   319→
   320→          updateJob({
   321→            processedBytes: result.size,
   322→            extractedBytes: result.size,
   323→            processedFiles: 1,
   324→          });
   325→
   326→          mainFileRef.current = file.name;
   327→        }
   328→
   329→        // Vyberieme main file pre detekciu
   330→        const mainFile = pickMainFile(extractedEntriesRef.current);
   331→        mainFileRef.current = mainFile;
   332→
   333→        // === DETECTING ===
   334→        updateJob({
   335→          status: "detecting",
   336→          currentStep: "Detekcia platformy...",
   337→        });
   338→
   339→        const detection = await runDetectionWorker({
   340→          fileName: mainFile,
   341→          fileSize: extractedEntriesRef.current.find((e) => e.path === mainFile)?.size ?? 0,
   342→        });
   343→        setDetectionResult(detection);
   344→
   345→        // === AWAITING USER SELECTION (ak potrebné) ===
   346→        if (detection.requiresUserSelection || !detection.platform) {
   347→          updateJob({
   348→            status: "awaiting-user-selection",
   349→            currentStep: "Vyberte platformu hry",
   350→          });
   351→          // Tu sa pipeline pozastaví — pokračuje sa až po kliknutí na tlačidlo
   352→          return;
   353→        }
   354→
   355→        // Pokračujeme priamo na finalization
   356→        await finalizeImport(detection.platform);
   357→
   358→         
   359→      } catch (e: any) {
   360→        if (abort.signal.aborted) {
   361→          updateJob({
   362→            status: "cancelled",
   363→            currentStep: "Import bol zrušený",
   364→          });
   365→          // Cleanup OPFS
   366→          await cleanupPartialImport(gameId).catch(() => undefined);
   367→          return;
   368→        }
   369→        const message = e instanceof Error ? e.message : String(e);
   370→        const code = e instanceof RetroCloudError ? e.code : "UNKNOWN_ERROR";
   371→        updateJob({
   372→          status: "error",
   373→          currentStep: "Import zlyhal",
   374→          error: `${message}${code !== "UNKNOWN_ERROR" ? ` (kód: ${code})` : ""}`,
   375→        });
   376→        // Cleanup OPFS pri chybe
   377→        await cleanupPartialImport(gameId).catch(() => undefined);
   378→        toast({
   379→          title: "Import zlyhal",
   380→          description: message,
   381→          variant: "destructive",
   382→        });
   383→      } finally {
   384→        if (workerRef.current) {
   385→          workerRef.current.terminate();
   386→          workerRef.current = null;
   387→        }
   388→      }
   389→    },
   390→     
   391→    [updateJob, currentJob, setJob, toast]
   392→  );
   393→
   394→  /**
   395→   * Spustí archive worker pre rozbalenie archívu.
   396→   * Vráti zoznam extrahovaných záznamov a súčasne streamuje dáta do OPFS.
   397→   */
   398→  const runArchiveWorker = useCallback(
   399→    (
   400→      format: "zip" | "rar" | "7z",
   401→      payload: File | ArrayBuffer,
   402→      fileName: string,
   403→      signal: AbortSignal
   404→    ): Promise<ArchiveEntry[]> => {
   405→      return new Promise<ArchiveEntry[]>((resolve, reject) => {
   406→        const worker = new Worker(
   407→          new URL("../../workers/archive.worker.ts", import.meta.url),
   408→          { type: "module" }
   409→        );
   410→        workerRef.current = worker;
   411→        const entries: ArchiveEntry[] = [];
   412→        let totalFiles = 0;
   413→        let totalBytes = 0;
   414→        let processedFiles = 0;
   415→        let processedBytes = 0;
   416→        let resolved = false;
   417→
   418→        const finish = (err?: Error) => {
   419→          if (resolved) return;
   420→          resolved = true;
   421→          worker.onmessage = null;
   422→          if (err) {
   423→            reject(err);
   424→          } else {
   425→            resolve(entries);
   426→          }
   427→        };
   428→
   429→        worker.onmessage = (e: MessageEvent<ArchiveWorkerResponse>) => {
   430→          const msg = e.data;
   431→          if (!msg || typeof msg !== "object" || !("type" in msg)) return;
   432→
   433→          switch (msg.type) {
   434→            case "progress": {
   435→              totalFiles = msg.totalFiles;
   436→              totalBytes = msg.totalBytes;
   437→              updateJob({
   438→                processedBytes: msg.processedBytes,
   439→                totalBytes,
   440→                processedFiles: msg.processedFiles,
   441→                fileCount: totalFiles,
   442→                extractedBytes: msg.processedBytes,
   443→              });
   444→              break;
   445→            }
   446→            case "warning": {
   447→              const warning = `${msg.path} — ${msg.reason}`;
   448→              updateJob({
   449→                warnings: [...(useImportStore.getState().currentJob?.warnings ?? []), warning],
   450→              });
   451→              break;
   452→            }
   453→            case "entry": {
   454→              // Pridáme do fronty na OPFS zápis
   455→              writeQueueRef.current.push({
   456→                path: msg.path,
   457→                data: msg.data,
   458→                size: msg.size,
   459→              });
   460→              entries.push({
   461→                path: msg.path,
   462→                size: msg.size,
   463→                isDirectory: false,
   464→              });
   465→              processedFiles += 1;
   466→              processedBytes += msg.size;
   467→              // Spustíme flush ak nie je aktívny
   468→              if (!writePromiseRef.current) {
   469→                writePromiseRef.current = flushWriteQueue(
   470→                  gameIdRef.current,
   471→                  abortRef.current?.signal ?? new AbortController().signal,
   472→                  (written) => {
   473→                    updateJob({ processedBytes: written });
   474→                  }
   475→                ).finally(() => {
   476→                  writePromiseRef.current = null;
   477→                });
   478→              }
   479→              break;
   480→            }
   481→            case "done": {
   482→              // Počkáme na vyprázdnenie write fronty
   483→              const wait = () => {
   484→                if (writePromiseRef.current) {
   485→                  writePromiseRef.current.then(wait, wait);
   486→                } else {
   487→                  finish();
   488→                }
   489→              };
   490→              wait();
   491→              break;
   492→            }
   493→            case "cancelled": {
   494→              finish(new RetroCloudError("UNKNOWN_ERROR", "Rozbaľovanie zrušené."));
   495→              break;
   496→            }
   497→            case "error": {
   498→              finish(
   499→                new RetroCloudError(
   500→                  (msg as { code: string }).code as RetroCloudError["code"],
   501→                  (msg as { message: string }).message
   502→                )
   503→              );
   504→              break;
   505→            }
   506→            case "entries":
   507→              // Ignorujeme — používame entry správy
   508→              break;
   509→            default:
   510→              // Neznáma správa — ignorujeme
   511→              break;
   512→          }
   513→        };
   514→
   515→        // Pošleme extract požiadavku
   516→        const request: ArchiveWorkerRequest = {
   517→          type: "extract",
   518→          format,
   519→          payload,
   520→          fileName,
   521→        };
   522→        // Pre ArrayBuffer použijeme transfer list
   523→        if (payload instanceof ArrayBuffer) {
   524→          worker.postMessage(request, [payload]);
   525→        } else {
   526→          worker.postMessage(request);
   527→        }
   528→
   529→        // Abort handler
   530→        signal.addEventListener("abort", () => {
   531→          if (workerRef.current) {
   532→            const cancelReq: ArchiveWorkerRequest = { type: "cancel" };
   533→            workerRef.current.postMessage(cancelReq);
   534→          }
   535→        });
   536→      });
   537→    },
   538→    [updateJob]
   539→  );
   540→
   541→  /**
   542→   * Postupne zapíše frontu extrahovaných záznamov do OPFS.
   543→   * Sériovo — aby sme nezaťažili OPFS paralelnými zápismi.
   544→   */
   545→  const flushWriteQueue = useCallback(
   546→    async (
   547→      gameId: string,
   548→      signal: AbortSignal,
   549→      onProgress: (written: number) => void
   550→    ): Promise<void> => {
   551→      let written = 0;
   552→      while (writeQueueRef.current.length > 0) {
   553→        if (signal.aborted) {
   554→          throw new RetroCloudError("UNKNOWN_ERROR", "Zápis zrušený.");
   555→        }
   556→        const item = writeQueueRef.current.shift()!;
   557→        const opfsPath = gameFilePath(gameId, item.path);
   558→        // Kopírujeme dáta do čistého ArrayBufferu, aby sme vyhli TS problémom
   559→        // s Uint8Array<ArrayBufferLike> v Blob konštruktore.
   560→        const buffer = new ArrayBuffer(item.data.byteLength);
   561→        new Uint8Array(buffer).set(item.data);
   562→        const blob = new Blob([buffer]);
   563→        const result = await streamBlobToOpfs(blob, opfsPath, (w) => {
   564→          onProgress(written + w);
   565→        }, signal);
   566→        written += result.size;
   567→        onProgress(written);
   568→      }
   569→    },
   570→    []
   571→  );
   572→
   573→  /**
   574→   * Spustí detection worker (mock).
   575→   */
   576→  const runDetectionWorker = useCallback(
   577→    async (input: { fileName: string; fileSize: number }): Promise<DetectionResult> => {
   578→      return new Promise<DetectionResult>((resolve, reject) => {
   579→        const worker = new Worker(
   580→          new URL("../../workers/detection.worker.ts", import.meta.url),
   581→          { type: "module" }
   582→        );
   583→        const id = uuid();
   584→        worker.onmessage = (e: MessageEvent) => {
   585→          const msg = e.data as { type: string; id: string; result?: DetectionResult; message?: string };
   586→          if (msg.type === "done" && msg.id === id) {
   587→            worker.terminate();
   588→            resolve(msg.result!);
   589→          } else if (msg.type === "error") {
   590→            worker.terminate();
   591→            reject(new Error(msg.message ?? "Detekcia zlyhala"));
   592→          }
   593→        };
   594→        worker.onerror = (e) => {
   595→          worker.terminate();
   596→          reject(new Error(`Worker error: ${e.message}`));
   597→        };
   598→        worker.postMessage({
   599→          type: "detect",
   600→          id,
   601→          input: {
   602→            fileName: input.fileName,
   603→            fileSize: input.fileSize,
   604→          },
   605→        });
   606→      });
   607→    },
   608→    []
   609→  );
   610→
   611→  /**
   612→   * Finalizuje import — uloží GameRecord + GameFileRecords do IndexedDB.
   613→   */
   614→  const finalizeImport = useCallback(
   615→    async (platform: EmulatorPlatform) => {
   616→      if (!currentJob) return;
   617→      const gameId = gameIdRef.current;
   618→      const entries = extractedEntriesRef.current;
   619→      const mainFile = mainFileRef.current || entries[0]?.path || "unknown";
   620→
   621→      updateJob({
   622→        status: "storing",
   623→        currentStep: "Ukladanie metadát...",
   624→      });
   625→
   626→      // Pre voľné súbory sme už streamovali — pre archívy sme streamovali počas extrakcie.
   627→      // Tu len uložíme metadáta.
   628→
   629→      // Vytvoríme GameFileRecord pre každý extrahovaný súbor
   630→      const fileRecords: GameFileRecord[] = entries
   631→        .filter((e) => !e.isDirectory)
   632→        .map((e) => {
   633→          const ext = extname(e.path);
   634→          return {
   635→            id: uuid(),
   636→            gameId,
   637→            relativePath: e.path,
   638→            opfsPath: gameFilePath(gameId, e.path),
   639→            fileName: e.path.split("/").pop() ?? e.path,
   640→            extension: ext,
   641→            size: e.size,
   642→          } satisfies GameFileRecord;
   643→        });
   644→
   645→      const totalSize = fileRecords.reduce((acc, f) => acc + f.size, 0);
   646→      const fingerprint = fingerprintGame(
   647→        fileRecords.map((f) => ({
   648→          relativePath: f.relativePath,
   649→          size: f.size,
   650→          hash: f.hash,
   651→        }))
   652→      );
   653→
   654→      const game: GameRecord = {
   655→        id: gameId,
   656→        name: deriveGameName(mainFile),
   657→        platform,
   658→        sourceType: currentJob.sourceType,
   659→        mainFile,
   660→        size: totalSize,
   661→        createdAt: Date.now(),
   662→        updatedAt: Date.now(),
   663→        totalPlayTimeSeconds: 0,
   664→        isFavorite: false,
   665→        compatibilityStatus: "unknown",
   666→        emulatorVersion: "v0.1.0",
   667→        fileFingerprint: fingerprint,
   668→      };
   669→
   670→      await putGame(game);
   671→      await Promise.all(fileRecords.map((f) => putGameFile(f)));
   672→      addGame(game);
   673→
   674→      updateJob({
   675→        status: "ready",
   676→        currentStep: "Import dokončený",
   677→        gameId,
   678→        processedFiles: fileRecords.length,
   679→        fileCount: fileRecords.length,
   680→      });
   681→
   682→      toast({
   683→        title: "Import dokončený",
   684→        description: `Hra "${game.name}" bola pridaná do knižnice.`,
   685→      });
   686→    },
   687→     
   688→    [currentJob, updateJob, addGame, toast]
   689→  );
   690→
   691→  /**
   692→   * Handler pre klik na tlačidlo "Pokračovať" po užívateľskej voľbe platformy.
   693→   */
   694→  const handleConfirmPlatform = useCallback(async () => {
   695→    if (!selectedPlatform) return;
   696→    try {
   697→      await finalizeImport(selectedPlatform);
   698→    } catch (e) {
   699→      const message = e instanceof Error ? e.message : String(e);
   700→      updateJob({
   701→        status: "error",
   702→        currentStep: "Import zlyhal",
   703→        error: message,
   704→      });
   705→    }
   706→  }, [selectedPlatform, finalizeImport, updateJob]);
   707→
   708→  /**
   709→   * Zrušenie importu.
   710→   */
   711→  const handleCancel = useCallback(() => {
   712→    abortRef.current?.abort();
   713→    if (workerRef.current) {
   714→      const cancelReq: ArchiveWorkerRequest = { type: "cancel" };
   715→      workerRef.current.postMessage(cancelReq);
   716→    }
   717→  }, []);
   718→
   719→  /**
   720→   * Reset wizardu — späť na začiatok.
   721→   */
   722→  const handleReset = useCallback(() => {
   723→    abortRef.current?.abort();
   724→    if (workerRef.current) {
   725→      workerRef.current.terminate();
   726→      workerRef.current = null;
   727→    }
   728→    reset();
   729→    setExtractedEntries([]);
   730→    setDetectionResult(null);
   731→    setSelectedPlatform(null);
   732→    extractedEntriesRef.current = [];
   733→    mainFileRef.current = "";
   734→    writeQueueRef.current = [];
   735→    writePromiseRef.current = null;
   736→  }, [reset]);
   737→
   738→  // === RENDER ===
   739→
   740→  // Právne potvrdenie pri prvom importe
   741→  if (!legalConfirmed) {
   742→    return (
   743→      <LegalConfirmation
   744→        checked={legalChecked}
   745→        onChange={setLegalChecked}
   746→        onConfirm={() => {
   747→          setLegalConfirmed(true);
   748→        }}
   749→      />
   750→    );
   751→  }
   752→
   753→  // Ak je job v programe (alebo dokončený), ukáž progress
   754→  if (currentJob && currentJob.status !== "idle" && currentJob.status !== "selecting") {
   755→    // Ak čaká na výber platformy, ukáž výber
   756→    if (currentJob.status === "awaiting-user-selection") {

   759→          detection={detectionResult}

   762→          onConfirm={handleConfirmPlatform}

   764→          fileName={currentJob.fileName}
   765→        />
   766→      );
   767→    }
   768→
   769→    return (
   770→      <div className="space-y-4">
   771→        <ImportProgress
   772→          job={currentJob}
   773→          onCancel={handleCancel}
   774→          onDismiss={
   775→            currentJob.status === "ready" || currentJob.status === "error" || currentJob.status === "cancelled"
   776→              ? handleReset
   777→              : undefined
   778→          }
   779→        />
   780→        {currentJob.status === "ready" && (
   781→          <Card>
   782→            <CardContent className="flex flex-col items-center gap-3 p-6 text-center">
   783→              <CheckCircle2 className="size-12 text-primary" aria-hidden="true" />
   784→              <div>
   785→                <p className="font-medium">Hra bola importovaná</p>
   786→                <p className="text-sm text-muted-foreground">
   787→                  {currentJob.fileName} — {currentJob.extractedBytes} bajtov
   788→                </p>
   789→              </div>
   790→              <div className="flex flex-col gap-2 sm:flex-row">
   791→                <Button asChild variant="default" className="min-h-11">
   792→                  <Link href="/library">
   793→                    <Package className="size-4" />
   794→                    Otvoriť knižnicu
   795→                  </Link>
   796→                </Button>
   797→                <Button variant="outline" onClick={handleReset} className="min-h-11">
   798→                  <RotateCcw className="size-4" />
   799→                  Importovať ďalšiu
   800→                </Button>
   801→              </div>
   802→            </CardContent>

   816→            <strong className="text-foreground">Tip:</strong> Pre DOS hry použite .jsdos
   817→            balík alebo ZIP s START.BAT / GAME.EXE. Pre PS1 použite .cue + .bin alebo .iso.
   818→          </p>
   819→        </CardContent>
   820→      </Card>
   821→    </div>
   822→  );
   823→}

   825→/**
   826→ * Vyberie "main file" hry z extrahovaných záznamov.
   827→ * Priorita: najväčší súbor s rozoznateľnou príponou.
   828→ */
   829→function pickMainFile(entries: ArchiveEntry[]): string {
   830→  const files = entries.filter(
   831→    (e) => !e.isDirectory && e.size > 0 && extname(e.path).length > 0
   832→  );
   833→  if (files.length === 0) return entries[0]?.path ?? "unknown";
   834→  files.sort((a, b) => b.size - a.size);
   835→  return files[0].path;
   836→}

   838→/**
   839→ * Odvodí názov hry z názvu súboru (bez prípony).
   840→ */
   841→function deriveGameName(mainFile: string): string {
   842→  const base = mainFile.split("/").pop() ?? mainFile;
   843→  const idx = base.lastIndexOf(".");
   844→  const name = idx === -1 ? base : base.slice(0, idx);
   845→  // Nahradíme podčiarkovníky a pomlčky medzerami
   846→  return name.replace(/[_-]+/g, " ").trim() || base;
   847→}
